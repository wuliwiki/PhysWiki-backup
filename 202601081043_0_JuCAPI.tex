\begin{itemize}
\item \href{https://docs.julialang.org/en/v1/manual/embedding/}{官方文档}。
\item \href{https://github.com/Clemapfel/jluna}{jluna} 是一个第三方 C++ wrapper。
\item 头文件：\verb`julia.h`，库文件： \verb`libjulia`
\begin{lstlisting}[language=C]
#include <julia.h>
JULIA_DEFINE_FAST_TLS // only define this once, in an executable
                     // (not in a shared library) if you want fast code.
int main(int argc, char *argv[])
{
    jl_init(); // 初始化

    jl_eval_string("print(sqrt(2.0))"); // 执行 julia 代码
    jl_value_t *ret = jl_eval_string("sqrt(2.0)"); // 返回 boxed value
    if (jl_typeis(ret, jl_float64_type)) { // 判断类型
        double ret_unboxed = jl_unbox_float64(ret); // 取出值
        printf("sqrt(2.0) in C: %e \n", ret_unboxed);
    }
    else
        printf("ERROR: unexpected type\n");

    jl_atexit_hook(0); // finalize
    return 0;
}
\end{lstlisting}
\item 编译命令：\verb`gcc -o test -fPIC -I$JULIA_DIR/include/julia -L$JULIA_DIR/lib -Wl,-rpath,$JULIA_DIR/lib test.c -ljulia`
\item 初始化函数： \verb`jl_init(), jl_init_with_image_file(), jl_init_with_image_handle()` 后者使用预编译 image。
\item \verb`jl_eval_string("print(sqrt(2.0))");` 执行字符串中的 julia 代码。
\end{itemize}
